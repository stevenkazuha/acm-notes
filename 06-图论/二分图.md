# 二分图与二分最大匹配

## 二分图判定

- **定义**：顶点分成两组，组内无边
- **判定**：染色法（0/1 交替染，出现相邻同色则不是二分图）

```cpp
vector<int> G[N];
int color[N];

bool dfs(int u, int c)
{
    color[u] = c;
    for(int v : G[u])
    {
        if(!color[v])
        {
            if(!dfs(v, 3-c)) return false;
        }
        else if(color[v] == c) return false;
    }
    return true;
}

bool is_bipartite(int n)
{
    memset(color, 0, sizeof(color));
    for(int i = 1; i <= n; i++)
        if(!color[i] && !dfs(i, 1)) return false;   // 不连通图逐块染色
    return true;
}
```

## 二分最大匹配（匈牙利算法）

```cpp
vector<int> G[N];
int match[N];      // 右边每个点匹配到左边谁
bool vis[N];

bool dfs(int u)
{
    for(int v : G[u])
    {
        if(!vis[v])
        {
            vis[v] = 1;
            if(!match[v] || dfs(match[v]))
            {
                match[v] = u;
                return true;
            }
        }
    }
    return false;
}

int hungary(int n)
{
    memset(match, 0, sizeof(match));
    int res = 0;
    for(int i = 1; i <= n; i++)
    {
        memset(vis, 0, sizeof(vis));
        if(dfs(i)) res++;
    }
    return res;
}
```
