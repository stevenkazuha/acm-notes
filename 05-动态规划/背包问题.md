# 背包问题

## 01 背包（每件物品选或不选）

```cpp
int v[N], w[N];      // 价值、体积
int dp[W+1];

void knapsack01(int n, int W)
{
    memset(dp, 0, sizeof(dp));
    for(int i = 1; i <= n; i++)
    {
        for(int j = W; j >= w[i]; j--)    // 倒序
        {
            dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
        }
    }
    cout << dp[W] << endl;
}
```

## 完全背包（每件物品可无限选）

```cpp
void knapsack_full(int n, int W)
{
    memset(dp, 0, sizeof(dp));
    for(int i = 1; i <= n; i++)
    {
        for(int j = w[i]; j <= W; j++)    // 正序
        {
            dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
        }
    }
    cout << dp[W] << endl;
}
```

## 多重背包（每件物品有数量限制 s[i]）

### 二进制优化

```cpp
int v[N], w[N], s[N];
int nv[N], nw[N], cnt;

void optimize(int n)
{
    cnt = 0;
    for(int i = 1; i <= n; i++)
    {
        int k = 1;
        while(k <= s[i])
        {
            nv[++cnt] = v[i] * k;
            nw[cnt] = w[i] * k;
            s[i] -= k;
            k *= 2;
        }
        if(s[i] > 0)
        {
            nv[++cnt] = v[i] * s[i];
            nw[cnt] = w[i] * s[i];
        }
    }
    // 转化为 01 背包
}
```

## 二维费用背包

```cpp
int dp[W+1][T+1];    // 体积 W，重量 T
for(int i = 1; i <= n; i++)
    for(int j = W; j >= w[i]; j--)
        for(int k = T; k >= t[i]; k--)
            dp[j][k] = max(dp[j][k], dp[j-w[i]][k-t[i]] + v[i]);
```

## 分组背包（每组选一个）

```cpp
vector<pair<int,int>> group[G];   // {价值, 体积}
int dp[W+1];

for(int g = 0; g < G; g++)
    for(int j = W; j >= 0; j--)
        for(auto& [v, w] : group[g])
            if(j >= w) dp[j] = max(dp[j], dp[j-w] + v);
```
