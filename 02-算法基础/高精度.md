# 高精度运算

## 什么时候需要高精度

- Python 天然支持任意大整数，C/C++ 不行
- 竞赛中当题目数据范围超过 `unsigned long long`（约 1.8×10^19）时必须上高精度
- 常见场景：阶乘（100! 有 158 位）、大数乘除、组合数 C(n, m)

---

## 先试 __int128

### 什么是 __int128

GCC/Clang 在 64 位平台提供的 128 位整数扩展，范围 ±2^127-1 ≈ **1.7×10^38**。MSVC 不支持，但绝大多数 OJ 都支持。

### 运算

直接加减乘除取模，跟普通 int 一模一样：

```cpp
__int128 a = 1234567890123456789LL;
__int128 b = 9876543210987654321LL;
__int128 c = a * b;          // 直接乘
__int128 d = a + b;          // 直接加
__int128 e = a % b;          // 直接取模
```

### 不能直接 cin/cout，要手写读写

```cpp
inline __int128 read()
{
    __int128 x = 0, f = 1;
    char ch = getchar();
    while(ch < '0' || ch > '9') { if(ch == '-') f = -1; ch = getchar(); }
    while(ch >= '0' && ch <= '9') { x = x * 10 + (ch - '0'); ch = getchar(); }
    return x * f;
}

inline void print(__int128 x)
{
    if(x < 0) { putchar('-'); x = -x; }
    if(x > 9) print(x / 10);
    putchar(x % 10 + '0');
}
```

### 什么时候用 __int128 vs 高精度 vs Python

| 范围 | 方案 |
|------|------|
| ≤ 1.8×10^19 | `unsigned long long` 够用 |
| **≤ 1.7×10^38** | **__int128**（首选，最方便） |
| > 1.7×10^38 或有其他操作 | Python 或 C++ 高精度 |

## 存储方式

### 倒序存

个位在下标 0：`1234` → `[4, 3, 2, 1]`

**为什么倒序？** 从低位开始运算，进位自然加到数组末尾，不需要移动整个数组。

### vector 实现（推荐）

- 动态大小，不会溢出
- 加减乘除都返回新 vector

## 高精度加法

### 复杂度

O(max(len(a), len(b)))

### 模板（vector 版）

```cpp
#include<vector>
#include<string>
using namespace std;

// A + B，两个都是倒序 vector<int>
vector<int> add(const vector<int>& A, const vector<int>& B)
{
    vector<int> C;
    int t = 0;
    for(int i = 0; i < A.size() || i < B.size() || t; i++)
    {
        if(i < A.size()) t += A[i];
        if(i < B.size()) t += B[i];
        C.push_back(t % 10);
        t /= 10;
    }
    return C;
}

// 使用示例
vector<int> A = {4, 3, 2, 1};    // 1234
vector<int> B = {8, 7, 6};       // 678
auto C = add(A, B);               // 1912 → {2, 1, 9, 1}
```

### 字符串版

```cpp
string add(string a, string b)
{
    vector<int> A, B;
    for(int i = a.size()-1; i >= 0; i--) A.push_back(a[i]-'0');
    for(int i = b.size()-1; i >= 0; i--) B.push_back(b[i]-'0');
    auto C = add(A, B);
    string res;
    for(int i = C.size()-1; i >= 0; i--) res += (char)('0'+C[i]);
    return res;
}
```

## 高精度减法

### 前提

A >= B（可以先比较大小，不够减加负号）

### 模板

```cpp
// 比较两个大整数（正整数，string 版）
bool cmp(const string& a, const string& b)
{
    if(a.size() != b.size()) return a.size() > b.size();
    return a >= b;   // 字符串比较字典序 = 数值大小（长度相同时）
}

vector<int> sub(const vector<int>& A, const vector<int>& B)
{
    vector<int> C;
    int t = 0;
    for(int i = 0; i < A.size(); i++)
    {
        t = A[i] - t;
        if(i < B.size()) t -= B[i];
        C.push_back((t + 10) % 10);
        t = (t < 0 ? 1 : 0);
    }
    // 去除前导零
    while(C.size() > 1 && C.back() == 0) C.pop_back();
    return C;
}

// 完整字符串版（自动处理负号）
string sub(string a, string b)
{
    bool neg = false;
    if(!cmp(a, b)) { swap(a, b); neg = true; }
    vector<int> A, B;
    for(int i = a.size()-1; i >= 0; i--) A.push_back(a[i]-'0');
    for(int i = b.size()-1; i >= 0; i--) B.push_back(b[i]-'0');
    auto C = sub(A, B);
    string res = neg ? "-" : "";
    for(int i = C.size()-1; i >= 0; i--) res += (char)('0'+C[i]);
    return res;
}
```

## 高精度乘低精度

### 复杂度

O(len(a))

### 模板

```cpp
vector<int> mul(const vector<int>& A, int b)
{
    vector<int> C;
    long long t = 0;
    for(int i = 0; i < A.size() || t; i++)
    {
        if(i < A.size()) t += (long long)A[i] * b;
        C.push_back(t % 10);
        t /= 10;
    }
    while(C.size() > 1 && C.back() == 0) C.pop_back();
    return C;
}
```

## 高精度乘高精度

### 朴素版 O(n·m)

```cpp
vector<int> mul(const vector<int>& A, const vector<int>& B)
{
    vector<int> C(A.size() + B.size(), 0);
    for(int i = 0; i < A.size(); i++)
        for(int j = 0; j < B.size(); j++)
            C[i + j] += A[i] * B[j];
    // 处理进位
    for(int i = 0; i < C.size()-1; i++)
    {
        C[i+1] += C[i] / 10;
        C[i] %= 10;
    }
    while(C.size() > 1 && C.back() == 0) C.pop_back();
    return C;
}
```

## 高精度除以低精度

### 复杂度

O(len(a))

### 关键点

除法要**正序**处理（从高位到低位），和加减法相反。

### 模板

```cpp
// A / b，返回商，r 存余数
vector<int> div(const vector<int>& A, int b, int& r)
{
    vector<int> C;
    long long t = 0;
    for(int i = A.size()-1; i >= 0; i--)
    {
        t = t * 10 + A[i];
        C.push_back(t / b);
        t %= b;
    }
    r = t;
    reverse(C.begin(), C.end());
    while(C.size() > 1 && C.back() == 0) C.pop_back();
    return C;
}
```

## 高精度完整模板

```cpp
#include<bits/stdc++.h>
using namespace std;

struct BigInt
{
    vector<int> d;
    bool neg = false;

    BigInt() {}
    BigInt(const string& s)
    {
        int i = 0;
        if(s[0] == '-') { neg = true; i = 1; }
        for(int j = s.size()-1; j >= i; j--) d.push_back(s[j]-'0');
    }
    BigInt(long long x)
    {
        if(x < 0) { neg = true; x = -x; }
        if(x == 0) d.push_back(0);
        while(x) { d.push_back(x%10); x /= 10; }
    }

    void trim()
    {
        while(d.size() > 1 && d.back() == 0) d.pop_back();
        if(d.size() == 1 && d[0] == 0) neg = false;
    }
    bool abs_less(const BigInt& o) const
    {
        if(d.size() != o.d.size()) return d.size() < o.d.size();
        for(int i = d.size()-1; i >= 0; i--)
            if(d[i] != o.d[i]) return d[i] < o.d[i];
        return false;
    }

    BigInt operator+(const BigInt& o) const
    {
        BigInt res;
        if(neg == o.neg)
        {
            res.neg = neg;
            int t = 0;
            for(int i = 0; i < d.size() || i < o.d.size() || t; i++)
            {
                if(i < d.size()) t += d[i];
                if(i < o.d.size()) t += o.d[i];
                res.d.push_back(t % 10);
                t /= 10;
            }
        }
        else
        {
            const BigInt* big = this;
            const BigInt* small = &o;
            if(abs_less(o)) { big = &o; small = this; }
            res.neg = big->neg;
            int t = 0;
            for(int i = 0; i < big->d.size(); i++)
            {
                t = big->d[i] - t;
                if(i < small->d.size()) t -= small->d[i];
                res.d.push_back((t + 10) % 10);
                t = (t < 0 ? 1 : 0);
            }
        }
        res.trim();
        return res;
    }

    BigInt operator-(const BigInt& o) const
    {
        BigInt tmp = o;
        tmp.neg = !tmp.neg;
        return *this + tmp;
    }

    BigInt operator*(int x) const
    {
        BigInt res;
        res.neg = neg ^ (x < 0);
        if(x < 0) x = -x;
        long long t = 0;
        for(int i = 0; i < d.size() || t; i++)
        {
            if(i < d.size()) t += (long long)d[i] * x;
            res.d.push_back(t % 10);
            t /= 10;
        }
        res.trim();
        return res;
    }

    BigInt operator/(int b) const
    {
        BigInt res;
        res.neg = neg ^ (b < 0);
        if(b < 0) b = -b;
        long long r = 0;
        for(int i = d.size()-1; i >= 0; i--)
        {
            r = r * 10 + d[i];
            res.d.push_back(r / b);
            r %= b;
        }
        reverse(res.d.begin(), res.d.end());
        res.trim();
        return res;
    }

    int operator%(int b) const
    {
        long long r = 0;
        for(int i = d.size()-1; i >= 0; i--)
            r = (r * 10 + d[i]) % b;
        return r;
    }

    friend ostream& operator<<(ostream& os, const BigInt& x)
    {
        if(x.neg) os << '-';
        for(int i = x.d.size()-1; i >= 0; i--) os << x.d[i];
        return os;
    }
};

// 使用
int main()
{
    BigInt a("123456789012345678901234567890");
    BigInt b("987654321098765432109876543210");
    cout << a + b << endl;
    cout << a - b << endl;
    cout << a * 5 << endl;
    cout << a / 3 << endl;
    return 0;
}
```

Python 的 `int` 原生支持**任意精度**，溢出自动变高精度。

### 什么时候用 Python vs C++

| 场景 | 选哪个 |
|------|--------|
| 大数加减乘除阶乘组合数 | **Python**（实现最短） |
| 大数取模 / 幂模 | **Python** 或 C++（都快） |
| unsigned long long 不够但 ≤ 1.7×10^38 | **__int128**（C++ 更方便） |
| 高精度 + 其他算法（线段树/图论） | **C++** |
| 模板未写完、临近超时 | 先交一版 Python 拿部分分 |

## 易错点

1. **C++ 前导零**：减法、乘法结果可能有，必须 `trim()`
2. **负号**：减法不够减要处理负号
3. **除法方向**：除法是正序处理，和加减法相反，容易写反
4. **数据范围**：高精度乘法中 `d[i] * b` 可能超过 int，用 `long long` 存中间结果
